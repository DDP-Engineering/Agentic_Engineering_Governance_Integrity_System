"""Compare frozen prompt-correction results to the selected previous current control."""
from pathlib import Path
import csv
import hashlib
import io
import json
import zipfile

root = Path(__file__).resolve().parents[1]
base = root / '05_Software/aegis/evaluations'
out = base / 'comparisons/2026-10-22-luna-prompt-corrections'
control = base / 'comparisons/2026-10-21-luna-original-current'
read = lambda p: json.loads(p.read_text('utf-8'))
dump = lambda p, v: p.write_text(json.dumps(v, indent=2, ensure_ascii=False), encoding='utf-8')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
rows = [dict(v, variant='baseline') for v in read(control / 'canonical-observations.json') if v['variant'] == 'current']
assert len(rows) == 25
control_sources = {}
for row in rows:
    directory = base / row['artifact_directory']
    original = next(r for r in read(directory / 'results.json') if r['scenario'] == row['scenario'])
    assert (original['passed'], original['status'], original['seconds'], original['output_tokens']) == (row['automatic_pass'], row['status'], row['seconds_including_nested_review'], row['coordinator_output_tokens'])
    control_sources[row['artifact_directory']] = read(directory / 'manifest.json')['sources']
dump(out / 'baseline-source-controls.json', control_sources)
reviews = []
archives = []
verified = []
protocol = read(out / 'protocol.json')
collection_seconds = 0
for profile in ['core', 'capabilities']:
    folder = base / 'candidates' / ('2026-10-22-luna-corrections-' + profile)
    manifest = read(folder / 'manifest.json')
    assert manifest['sources'] == protocol['harness_and_roles_unchanged']
    assert manifest['payload_files'] == protocol['candidate_payload']
    assert manifest['cli_sha256'] == protocol['cli_sha256']
    assert (manifest['model'], manifest['reasoning'], manifest['context']) == ('gpt-6-luna', 'medium', 131072)
    for name, expected in manifest['sources'].items():
        assert sha(root / name) == expected, name
    work = root / '.tmp-codex' / ('luna-corrections-' + profile)
    results = read(folder / 'results.json')
    assert len(results) == (10 if profile == 'core' else 15)
    collection_seconds += read(folder / 'timing.json')['collection_wall_seconds']
    for item in results:
        run = work / item['run_directory']
        tools = read(run / 'coordinator/tools.json')
        chunks = [t for t in tools if t['tool'] == 'read_file' and t['arguments'].get('path', '').replace('\\', '/') == '.agentic/AGENTS.md' and 'text' in t['result']]
        covered, total = set(), 0
        for t in chunks:
            start = max(0, int(t['arguments'].get('offset', 0)))
            covered.update(range(start, start + len(t['result']['text'])))
            total = max(total, t['result'].get('total_chars', 0))
        journals = list((run / 'repo/.agentic/tasks').glob('TASK-*.md'))
        row = {'variant': 'corrected', 'profile': profile, 'scenario': item['scenario'],
            'collection': '2026-10-22', 'artifact_directory': folder.relative_to(base).as_posix(),
            'model': manifest['model'], 'reasoning': manifest['reasoning'], 'configured_context_tokens': manifest['context'],
            'native_output_cap_tokens': 'unexposed', 'status': item['status'], 'automatic_pass': item['passed'],
            'seconds_including_nested_review': item['seconds'], 'coordinator_output_tokens': item['output_tokens'],
            'reviewer_output_tokens': item['reviewer_output_tokens'], 'reviewer_sessions': item['reviewer_sessions'],
            'coordinator_reasoning_tokens': item['reasoning_tokens'], 'reviewer_reasoning_tokens': item['reviewer_reasoning_tokens'],
            'failed_checks': '; '.join(c['id'] for c in item['evaluation']['checks'] if not c['passed']),
            'agents_contract_read': bool(chunks), 'agents_contract_complete_read': bool(total) and len(covered) >= total,
            'step_prompts_read': '; '.join(sorted({t['arguments']['path'] for t in tools if t['tool'] == 'read_file' and t['arguments'].get('path', '').startswith('.agentic/steps/') and 'text' in t['result']})),
            'journal_license_prologue': len(journals) == 1 and all(s in journals[0].read_text('utf-8')[:650] for s in ['Copyright (c) 2026 DDP Engineering', 'LicenseRef-PolyForm-Noncommercial-1.0.0'])}
        rows.append(row)
        ledger = run / 'coordinator/reviews.json'
        if ledger.exists():
            for review in read(ledger):
                reviews.append({'variant': 'corrected', 'scenario': item['scenario'],
                    'model': manifest['model'], 'reasoning': manifest['reasoning'], 'configured_context_tokens': manifest['context'],
                    'native_output_cap_tokens': None, **review})
    with zipfile.ZipFile(folder / 'evidence.zip') as archive:
        assert archive.testzip() is None
        archives.append({'profile': profile, 'crc_ok': True, 'entries': len(archive.namelist())})
    for name, expected in read(folder / 'integrity.json').items():
        assert sha(folder / name) == expected, name
        verified.append({'path': str((folder / name).relative_to(base)), 'sha256': expected})
for name, expected in protocol['candidate_payload'].items():
    assert sha(root / '05_Software/aegis/template/.agentic' / name) == expected
    assert sha(root / '.agentic' / name) == expected
assert len(rows) == 50
dump(out / 'observations.json', rows)
dump(out / 'reviewer-sessions.json', reviews)
dump(out / 'archive-checks.json', archives)
dump(out / 'verified-integrity.json', verified)
stream = io.StringIO(newline='')
writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
writer.writeheader()
writer.writerows(rows)
(out / 'observations.csv').write_text(stream.getvalue(), encoding='utf-8', newline='')
counts = {v: {p: sum(r['automatic_pass'] for r in rows if r['variant'] == v and r['profile'] == p) for p in ['core', 'capabilities']} for v in ['baseline', 'corrected']}
metrics = {'strict_scores': counts, 'candidate_collection_seconds': collection_seconds,
    'candidate_coordinator_cases': 25, 'candidate_fresh_reviewer_sessions': len(reviews),
    'case_totals': {v: {'seconds_including_review': sum(r['seconds_including_nested_review'] for r in rows if r['variant'] == v),
        'output_tokens': sum(r['coordinator_output_tokens'] + r['reviewer_output_tokens'] for r in rows if r['variant'] == v),
        'completed': sum(r['status'] == 'completed' for r in rows if r['variant'] == v)} for v in ['baseline', 'corrected']}}
dump(out / 'metrics.json', metrics)
lines = ['<!-- Copyright (c) 2026 DDP Engineering. LicenseRef-PolyForm-Noncommercial-1.0.0 -->', '',
    '# Prompt corrections compared with the preceding current baseline', '',
    'The same 25 cases and frozen scorer compare the corrected payload with the selected current control from 2026-10-21. The control preserves the symmetrically repaired review subgroup. No semantic failures were retried.', '',
    '| Setting | Value |', '|---|---|', '| Model | `gpt-6-luna` |', '| Reasoning | `medium` |',
    '| Configured context | 131,072 tokens (128K); server allocation not independently exposed |',
    '| Native output cap | Unexposed; 16,000 is the MCP tool-output allowance |',
    '| Repeats | One per case; historical matched control, not interleaved |',
    '| Pacing / returned 429 backoff | 30 seconds before each session; 60/120/240 seconds |',
    '| Runtime / harness / role prompts | Fixed since the declared baseline transport repair; exact source hashes recorded |',
    '| Payload version | 0.14.0 |', '',
    'Only AEGIS prompt material changed. [Protocol and exact hashes](protocol.json) and [the correction map](changes.md) identify the payload edits; Frame, Build, the parser, scenarios, harness and native role definitions are unchanged since the declared transport repair. Installed and template payload bytes match the frozen candidate. The previous control retains its original non-review executions: their older adapter differs only in the unreachable review-reference branch, as documented in the prior baseline. Review controls use the repaired adapter. [Control source manifests](baseline-source-controls.json) preserve that distinction.', '',
    '| Profile | Previous current | Corrected |', '|---|---:|---:|']
for profile, size in [('core', 10), ('capabilities', 15)]:
    lines.append(f"| {profile} | {counts['baseline'][profile]}/{size} | {counts['corrected'][profile]}/{size} |")
lines += [f"| Total | {sum(counts['baseline'].values())}/25 | {sum(counts['corrected'].values())}/25 |", '',
    'Strict predicates are not a complete semantic quality score. Read [the evidence-based semantic assessment](semantic-review.json); known narrow predicates remain unchanged. One repeat does not establish a universal success rate or isolate individual prompt contributions.', '',
    '## Per-case results', '',
    'Output includes coordinator and nested fresh reviewer output. Wall time already includes nested review; pre-case pauses are excluded. Model, reasoning, configured context, native output-cap status and token usage are repeated in [JSON](observations.json) and [CSV](observations.csv).', '',
    '| Scenario | Previous | Corrected | Seconds P/C | Output tokens P/C | Failed candidate checks |',
    '|---|---|---|---:|---:|---|']
for before in rows[:25]:
    after = next(r for r in rows[25:] if r['scenario'] == before['scenario'])
    state = lambda r: 'pass' if r['automatic_pass'] else 'fail'
    tokens = lambda r: r['coordinator_output_tokens'] + r['reviewer_output_tokens']
    lines.append(f"| `{before['scenario']}` | {state(before)} | {state(after)} | {before['seconds_including_nested_review']:.1f} / {after['seconds_including_nested_review']:.1f} | {tokens(before):,} / {tokens(after):,} | {after['failed_checks'] or '—'} |")
lines += ['', '## Observed time and usage', '', '| Variant | Case wall seconds, including reviewers | Output tokens |', '|---|---:|---:|']
for variant, totals in metrics['case_totals'].items():
    lines.append(f"| {variant} | {totals['seconds_including_review']:.1f} | {totals['output_tokens']:,} |")
lines += ['', f"Candidate collection including pauses took {collection_seconds:.1f} seconds and launched {len(reviews)} fresh independent reviewer sessions. Actual reviewer records include session IDs, digests, verdicts, timing and observed reads in [reviewer sessions](reviewer-sessions.json).", '',
    'Time and output are observations, not an isolated efficiency claim: runs occurred on different dates with one repeat and native cache/service variation. Cumulative input tokens span multiple turns and are not peak context occupancy.', '',
    '## Verification and scope limits', '',
    'The same bounded fixture capabilities apply: eight step prompts plus the shared AGENTS/task contract; independent prepared stages, not one continuous project. Build permits one pure-function correction. Deliver stops before publication. Adoption covers one agreed component. Diagrams are inspected as saved Mermaid artifacts; rendering is not certified. Native installed-plugin discovery is not exercised. Reviewer approval/fix-loop coverage depends on the actual verdicts.', '',
    'Both raw archives passed CRC and [recorded hashes were verified](verified-integrity.json). [Software verification](verification.md) records local checks. No own AEGIS workflow, version bump, commit or publication was performed.', '',
    '## Raw collections', '',
    '- [Core results](../../candidates/2026-10-22-luna-corrections-core/results.json), [manifest](../../candidates/2026-10-22-luna-corrections-core/manifest.json), [evidence](../../candidates/2026-10-22-luna-corrections-core/evidence.zip).',
    '- [Capability results](../../candidates/2026-10-22-luna-corrections-capabilities/results.json), [manifest](../../candidates/2026-10-22-luna-corrections-capabilities/manifest.json), [evidence](../../candidates/2026-10-22-luna-corrections-capabilities/evidence.zip).',
    '- [Previous baseline and semantic assessment](../2026-10-21-luna-original-current/report.md).', '',
    'General prompting guidance consulted: [OpenAI prompt engineering](https://developers.openai.com/api/docs/guides/prompt-engineering). Repository evidence and unchanged scenario outcomes govern this comparison.', '']
(out / 'report.md').write_text('\n'.join(lines), encoding='utf-8')
print(json.dumps(metrics, indent=2))
