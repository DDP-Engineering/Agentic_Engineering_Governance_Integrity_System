"""Apply the user's external-capacity exclusion with a grade-independent selector."""
from pathlib import Path
import copy
import csv
import hashlib
import io
import json
import zipfile

root=Path(__file__).resolve().parents[1]
base=root/'05_Software/aegis/evaluations'
out=base/'comparisons/2026-10-22-luna-validation-records'
read=lambda p:json.loads(p.read_text('utf-8'))
dump=lambda p,v:p.write_text(json.dumps(v,indent=2,ensure_ascii=False),encoding='utf-8')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
protocol=read(out/'protocol.json')
attempts=read(out/'operational-recovery-results.json')
chosen=next(a for a in attempts if a['coordinator_status']=='completed' and not any(s in ['provider_error','run_timeout'] for s in a['role_statuses'].values()))
folder=base/chosen['output']
manifest=read(folder/'manifest.json')
assert manifest['sources']==protocol['harness_and_roles_unchanged']
assert manifest['payload_files']==protocol['candidate_payload']
assert (manifest['model'],manifest['reasoning'],manifest['context'])==('gpt-6-luna','medium',131072)
assert manifest['cli_sha256']==protocol['cli_sha256']
result=read(folder/'results.json')[0]
work=root/'.tmp-codex'/('review-capacity-diagnostic-'+str(chosen['attempt']))/result['run_directory']
reviews=read(work/'coordinator/reviews.json')
assert len(reviews)==2 and all(r['status']=='completed' for r in reviews)
rows=copy.deepcopy(read(out/'observations.json'))
row=next(r for r in rows if r['variant']=='corrected' and r['scenario']==result['scenario'])
excluded=copy.deepcopy(row)
row.update(collection='capacity-recovery-first-operationally-complete',artifact_directory=folder.relative_to(base).as_posix(),
    status=result['status'],automatic_pass=result['passed'],seconds_including_nested_review=result['seconds'],
    coordinator_output_tokens=result['output_tokens'],reviewer_output_tokens=result['reviewer_output_tokens'],
    reviewer_sessions=result['reviewer_sessions'],coordinator_reasoning_tokens=result['reasoning_tokens'],
    reviewer_reasoning_tokens=result['reviewer_reasoning_tokens'],
    failed_checks='; '.join(c['id'] for c in result['evaluation']['checks'] if not c['passed']))
tools=read(work/'coordinator/tools.json')
chunks=[e for e in tools if e['tool']=='read_file' and e['arguments'].get('path')=='.agentic/AGENTS.md' and 'text' in e['result']]
covered,total=set(),0
for e in chunks:
    offset=int(e['arguments'].get('offset',0));covered.update(range(offset,offset+len(e['result']['text'])));total=max(total,e['result'].get('total_chars',0))
row.update(agents_contract_read=bool(chunks),agents_contract_complete_read=bool(total) and len(covered)>=total,
    step_prompts_read='; '.join(sorted({e['arguments']['path'] for e in tools if e['tool']=='read_file' and e['arguments'].get('path','').startswith('.agentic/steps/') and 'text' in e['result']})))
text=(work/'repo/.agentic/tasks/TASK-aaaaaaaa.md').read_text('utf-8')
row['journal_license_prologue']=all(s in text[:650] for s in ['Copyright (c) 2026 DDP Engineering','LicenseRef-PolyForm-Noncommercial-1.0.0'])
dump(out/'canonical-observations.json',rows)
stream=io.StringIO(newline='');writer=csv.DictWriter(stream,fieldnames=list(rows[0]));writer.writeheader();writer.writerows(rows)
(out/'canonical-observations.csv').write_text(stream.getvalue(),encoding='utf-8',newline='')
canonical_reviews=[r for r in read(out/'reviewer-sessions.json') if r['scenario']!=result['scenario']]
canonical_reviews.extend({'variant':'corrected','scenario':result['scenario'],'model':'gpt-6-luna','reasoning':'medium','configured_context_tokens':131072,'native_output_cap_tokens':None,**r} for r in reviews)
dump(out/'canonical-reviewer-sessions.json',canonical_reviews)
dump(out/'excluded-external-observations.json',{'reason':'Native Spec role rejected by external ChatGPT capacity; excluded per user instruction.','observation':excluded,'independent_semantic_finding_retained':'Extra task journal was created before provider rejection. This observation is not erased by selection of a valid independent follow-up.'})
selection={'user_instruction':'ChatGPT capacity is external and must not count against prompt quality.',
    'rule':'Select the first operationally complete follow-up regardless of semantic grade. Preserve raw grades and independent semantic findings.',
    'selected':chosen,'original_raw_observations_preserved':'observations.json',
    'canonical_cases_per_variant':25,'canonical_reviewer_attempts':len(canonical_reviews),
    'canonical_reviewers_completed':sum(r['status']=='completed' for r in canonical_reviews),
    'failed_original_role_error':'Selected model is at capacity','initial_duplicate_journal_finding_retained':True}
dump(out/'selection.json',selection)
verified=read(out/'verified-integrity.json')
for name,expected in read(folder/'integrity.json').items():
    assert sha(folder/name)==expected,name
    verified.append({'path':str((folder/name).relative_to(base)),'sha256':expected})
dump(out/'verified-integrity.json',verified)
archives=read(out/'archive-checks.json')
with zipfile.ZipFile(folder/'evidence.zip') as archive:
    assert archive.testzip() is None
    archives.append({'profile':'capacity-diagnostic','crc_ok':True,'entries':len(archive.namelist())})
dump(out/'archive-checks.json',archives)
metrics=read(out/'metrics.json')
metrics['raw_cohort_strict_scores']=metrics['strict_scores']
metrics['strict_scores']={v:{p:sum(r['automatic_pass'] for r in rows if r['variant']==v and r['profile']==p) for p in ['core','capabilities']} for v in ['baseline','corrected']}
metrics['case_totals']={v:{'seconds_including_review':sum(r['seconds_including_nested_review'] for r in rows if r['variant']==v),'output_tokens':sum(r['coordinator_output_tokens']+r['reviewer_output_tokens'] for r in rows if r['variant']==v),'completed':sum(r['status']=='completed' for r in rows if r['variant']==v)} for v in ['baseline','corrected']}
metrics['physical_second_cohort_and_recovery_seconds']=metrics['candidate_collection_seconds']+sum(read(base/a['output']/'timing.json')['collection_wall_seconds'] for a in attempts)
metrics['physical_second_cohort_and_recovery_coordinators']=25+len(attempts)
metrics['physical_second_cohort_and_recovery_role_attempts']=len(read(out/'reviewer-sessions.json'))+len(reviews)
metrics['canonical_reviewer_statuses']={s:sum(r['status']==s for r in canonical_reviews) for s in {r['status'] for r in canonical_reviews}}
metrics['external_capacity_excluded']=True
dump(out/'metrics.json',metrics)
notes=read(out/'semantic-review.json')
case=next(c for c in notes['cases'] if c['scenario']==result['scenario'])
case['excluded_cohort_assessment']=case['assessment']
case['assessment']='Canonical follow-up has both role sessions completed, digest checked twice by both and snapshot unchanged. Quality correctly identifies None/ValueError missing. Spec incorrectly treats untracked files absent from HEAD as unavailable current scope, instead of reading them through available file tools. Strict independent_read_evidence fails for insufficient actual Spec inspection; this is a semantic/tool-use failure, not provider capacity. One original journal is retained in this follow-up, actual blocked verdicts are saved and no repair/final proof occurs. Final summary preserves truncation warning but is unfenced. Initial excluded run still independently demonstrated duplicate-journal creation before capacity rejection.'
notes['external_capacity_selection']='selection.json; capacity-contaminated observation is not scored.'
dump(out/'semantic-review.json',notes)
print(json.dumps({'canonical_scores':metrics['strict_scores'],'canonical_reviewers':metrics['canonical_reviewer_statuses'],'selected_checks_failed':row['failed_checks']},indent=2))
